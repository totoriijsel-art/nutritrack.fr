async function searchFood(q) { if (searchCache[q]) { displayResults(searchCache[q]); return; } try { const url = `https://world.openfoodfacts.org/cgi/search.pl?search_terms=${encodeURIComponent(q)}&search_simple=1&action=process&json=1&page_size=30`; const res = await fetch(url); const data = await res.json(); let qLower = q.toLowerCase(); let products = (data.products || []).map(p => ({ name: cleanFoodName(p.product_name_fr || p.product_name || 'Inconnu'), calories: p.nutriments?.['energy-kcal_100g'] || p.nutriments?.['energy-kcal'] || 0, proteins: p.nutriments?.['proteins_100g'] || 0, carbohydrates: p.nutriments?.['carbohydrates_100g'] || 0, fat: p.nutriments?.['fat_100g'] || 0 })).filter(p => { if (!p.name || p.name === 'Inconnu') return false; let nameLower = p.name.toLowerCase(); // Si on recherche un fruit brut, on écarte les produits transformés évidents if (qLower.length <= 6) { const bannedWords = ['confiture', 'danino', 'yaourt', 'sirop', 'biscuits', 'gâteau', 'glace', 'tarte', 'boisson', 'nectar', 'jus', 'arôme', 'préparation', 'carré frais', 'jben']; for (let word of bannedWords) { if (nameLower.includes(word)) return false; } } return true; }); // Tri strict : met les correspondances exactes ou les noms les plus courts (fruits bruts) en premier products.sort((a, b) => { let aName = a.name.toLowerCase(); let bName = b.name.toLowerCase(); let aExact = aName === qLower ? 0 : 1; let bExact = bName === qLower ? 0 : 1; if (aExact !== bExact) return aExact - bExact; return aName.length - bName.length; }); searchCache[q] = products; localStorage.setItem('searchCache', JSON.stringify(searchCache)); displayResults(products); } catch (e) { console.error('Erreur API Open Food Facts:', e); } }
